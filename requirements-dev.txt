# ======================================================================
# 4. LINTING / CODE QUALITY
# ======================================================================
#
# Ruff provides fast static checks for:
#
#   unused imports
#   undefined names
#   suspicious code
#   import problems
#   common Python mistakes
#   style consistency
#
#
# Example:
#
#   ruff check .
#
#
# Automatic safe fixes:
#
#   ruff check . --fix
#
#
# Formatting can also later be enabled through:
#
#   ruff format .
#
#
# Final rule selection will be centralized in pyproject.toml.
# ======================================================================

ruff>=0.8,<1.0


# ======================================================================
# 5. STATIC TYPE CHECKING
# ======================================================================
#
# mypy provides static checking for the strongly typed parts of the
# Python architecture.
#
# Particularly valuable for:
#
#   protocol.py
#   node.py
#   stream_manager.py
#   localization/
#   calibration/
#   classification/
#   analytics/
#   database interfaces
#
#
# Example:
#
#   mypy .
#
#
# Third-party scientific libraries do not always expose perfect type
# information, so the final mypy policy will initially remain practical
# rather than maximally strict.
# ======================================================================

mypy>=1.11,<2.0
